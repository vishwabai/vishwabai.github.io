# Class definition — the blueprint
class Shirt:
  def __init__(self, size, color, fabric):
    self.size = size        # attribute
    self.color = color
    self.is_worn = False

  def wear(self):         # method
    self.is_worn = True

# Objects — the real garments
shirt1 = Shirt("M", "white", "cotton")
shirt2 = Shirt("XL", "navy", "linen")

Encapsulation

Bundle data (attributes) and methods inside the garment. Keep internal details private — expose only what the outside world needs.

class Pants:
  def __init__(self, size, color):
    self.size = size                # public
    self.__stitching_cost = 12.5  # private

  def get_size(self):             # public getter
    return self.size

# Outside — can access public, not private
p = Pants("32", "black")
p.get_size()                    # ✅ "32"
p.__stitching_cost             # ❌ AttributeError!

class Clothing:                     # parent
  def __init__(self, size, color):
    self.size = size
  def wear(self): ...

class Shirt(Clothing):                 # child
  def __init__(self, size, color, collar):
    super().__init__(size, color)   # inherit!
    self.collar_type = collar

class FormalShirt(Shirt):             # grandchild
  tie_compatible = True

Concept 04

Polymorphism

Same method name, different behavior. Wearing a shirt and wearing pants both use wear() — but the experience is unique to each garment.

SHIRT.wear()

Shirt

PANTS.wear()

Pants

JACKET.wear()

Jacket

# Same method name — different behaviour!
class Shirt(Clothing):
  def wear(self): return "👔 Tucking shirt in..."

class Pants(Clothing):
  def wear(self): return "👖 Stepping into pants..."

class Jacket(Clothing):
  def wear(self): return "🧥 Slipping jacket on..."

# Polymorphism in action:
for item in [Shirt(), Pants(), Jacket()]:
  print(item.wear())  # same call, unique output

Concept 05

Abstraction

Define a common contract every clothing item must fulfil. You don't care how each garment is worn — just that it can be worn.

ABSTRACT CLASS

Clothing

Every garment MUST implement:
wear()
remove()
get_description()
Cannot be instantiated directly!
→
✅ Shirt implements

wear() → "Puts on shirt"
remove() → "Takes off shirt"
get_description() → "Cotton M shirt"

✅ Pants implements

wear() → "Steps into pants"
remove() → "Steps out"
get_description() → "Linen 32 pants"

from abc import ABC, abstractmethod

class Clothing(ABC):              # abstract
  @abstractmethod
  def wear(self): pass

  @abstractmethod
  def get_description(self): pass

class Shirt(Clothing):             # concrete
  def wear(self): return "Shirt on!"
  def get_description(self): return "Cotton shirt"
