01Create & read data

pd.Series([1, 2, 3]) # 1-D labeled array
pd.DataFrame({"a": [1,2], "b": [3,4]}) # dict of columns
pd.read_csv("f.csv", parse_dates=["d"])
pd.read_csv("f.csv", usecols=["a","b"], dtype={"a":"int32"})
pd.read_excel("f.xlsx", sheet_name=0)
pd.read_json("f.json")
pd.read_sql(query, conn)
df.to_csv("out.csv", index=False)

02Inspect

df.head(n) / df.tail(n)
df.shape # (rows, cols)
df.info() # dtypes + non-null counts
df.describe() # numeric summary stats
df.dtypes / df.columns / df.index
df.isna().sum() # missing values per column
df["c"].nunique() / .value_counts()
df.sample(5) # random rows

03Select & filter

df["col"] / df[["a","b"]]
df.loc[rows, cols] # label-based
df.iloc[rows, cols] # position-based
df[df["a"] > 5] # boolean mask
df[(df.a > 5) & (df.b == "x")] # use & | ~, not and/or
df.query("a > 5 and b == 'x'")
df.at[row, "col"] # fast single scalar

!loc vs iloc

df.loc[0:2] # label-based, END INCLUSIVE
df.iloc[0:2] # position-based, end exclusive
df["a"][df.a > 0] = 0 # chained assignment — avoid
df.loc[df.a > 0, "a"] = 0 # correct way

04Clean & missing data

df.isna() / df.notna()
df.fillna(value)
df.groupby("g")["a"].transform(lambda s: s.fillna(s.mean()))
df.dropna(subset=["a"])
df.duplicated(subset=["id"])
df.drop_duplicates(subset=["id"], keep="first")
df.replace({"old": "new"})
df.astype({"a": "int64"})

05Transform columns

df["c"] = df["a"] + df["b"] # vectorized
df.assign(c=lambda d: d.a * 2)
df["c"].apply(func) # row-wise, slower
df.apply(func, axis=1)
np.where(cond, x, y) # vectorized if/else
df.rename(columns={"a": "A"})
df.set_index("id") / df.reset_index()
df.pipe(func) # chain a custom function

06Sort & rank

df.sort_values("a", ascending=False)
df.sort_values(["a","b"], ascending=[True, False])
df.nlargest(n, "a") / df.nsmallest(n, "a")
df["a"].rank(method="dense")
df.sort_index()

07GroupBy & aggregate

df.groupby("g").agg(x=("a","mean"), y=("b","sum"))
df.groupby("g")["a"].sum()
df.groupby("g").size() # row count per group
df.groupby("g")["a"].transform("mean") # same shape as df
df.groupby("g").filter(lambda g: len(g) > 2)
df.groupby("g").apply(func) # full flexibility, slowest
df.groupby(["g1","g2"]).agg(...) # → MultiIndex

08Merge & concat

pd.merge(a, b, on="key", how="left")
pd.merge(a, b, left_on="x", right_on="y")
how: "inner" | "left" | "right" | "outer"
pd.concat([a, b], ignore_index=True) # stack rows
pd.concat([a, b], axis=1) # stack columns
a.join(b, on="key")

09Reshape

df.pivot_table(values="v", index="i", columns="c", aggfunc="mean")
df.melt(id_vars=["id"], var_name="k", value_name="v")
df.stack() / df.unstack()
df.T # transpose
pd.crosstab(df.a, df.b)

10String methods (.str)

df["s"].str.lower() / .str.upper()
df["s"].str.contains("x", na=False)
df["s"].str.replace("a", "b")
df["s"].str.split(",").str[0]
df["s"].str.strip() / .str.len()
df["s"].str.extract(r"(\d+)")

11Dates & time series

pd.to_datetime(df["d"])
pd.date_range("2026-01-01", periods=30, freq="D")
df.set_index("date").resample("W").sum()
df["a"].rolling(7).mean() # moving average
df["a"].shift(1) # lag / .shift(-1) # lead
df["a"].ewm(span=7).mean() # exp. weighted
df["date"].dt.year / .dt.dayofweek / .dt.is_month_end

12MultiIndex

df.set_index(["a","b"])
df.groupby(["a","b"]).agg(...)
df.loc[("x","y")] # select by full tuple key
df.xs("x", level="a") # select one level
df.reset_index() # flatten back to columns

13Categorical & memory

df["c"] = df["c"].astype("category") # low-cardinality cols
df.memory_usage(deep=True)
pd.cut(df["a"], bins=[0,10,50], labels=["lo","hi"])
pd.qcut(df["a"], q=4) # equal-sized buckets

14Plotting

df["a"].plot() # line, datetime index → time axis
df.plot(kind="bar" | "barh" | "hist" | "box")
df.groupby("g")["a"].mean().plot(kind="bar")
df.plot.scatter(x="a", y="b")

!Performance

df.query("a > 5") # often faster than a mask
df.eval("c = a + b")
np.where(...) over df.apply(lambda...) # 10-100x faster
avoid df.iterrows() / itertuples() in loops
pd.read_csv("big.csv", chunksize=100_000) # stream
df["c"].astype("category") # cuts memory + groupby time

15Common ML feature patterns

df.groupby("id").agg(recency=("date", lambda d: (today-d.max()).days), frequency=("date","count"), monetary=("amt","sum"))
df["lag_1"] = df["a"].shift(1) # forecasting feature
train = df[df.index < cutoff_date] # time-based split, no leakage
scored = preds.merge(df, on="id") scored.groupby("segment")["error"].mean() # error by segment
df.values / df.to_numpy() # hand off to sklearn/torch

16Cumulative & expanding

df["a"].cumsum() / .cumprod() / .cummax() / .cummin()

df["a"].diff() # row-to-row change
df["a"].pct_change() # % change vs previous row

17Index alignment & ops

df1 + df2 # aligns on index automatically
df.reindex(new_index)
df.align(other, join="outer")

20Conditional assignment

df["a"].where(cond, other) # keep if True, else replace
df["a"].mask(cond, other) # opposite of .where()
df["a"].combine_first(df["b"]) # fill nulls from another col
np.select([c1, c2], [v1, v2], default=v3) # multi-branch

21Display & options

22Testing & equality

df.equals(other) # True/False, exact match
pd.testing.assert_frame_equal(a, b) # for unit tests
df["a"].isin([1, 2, 3])

!More gotchas

df_copy = df.copy() # avoid SettingWithCopyWarning
df[mask]["a"] = 1 # WRONG — chained, may silently fail
df.loc[mask, "a"] = 1 # correct
0.1 + 0.2 == 0.3 # False — use np.isclose() for floats
df.empty # True if DataFrame has 0 rows
